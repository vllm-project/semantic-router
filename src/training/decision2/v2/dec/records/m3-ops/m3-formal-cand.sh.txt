set -u
# Decoder M3 prereg rule 3 (5c8bbc569): one finalist on node A — download the staged folder at its
# commit, verify every file against node B's SHA-256 list, collect once at 16,384 tokens (smoke first),
# score vs the adopted 1.0 run, the 16K same-limit control and the tier's best peer, then mlx-diag.
# usage: m3-formal-cand.sh <mirror-dir> <name e.g. N4T-soup> <staging commit> <tier 2B|4B|0.8B> <source 1.0 package> <adopted run> <16K control run>
SRC=$1 NAME=$2 COMMIT=$3 TIER=$4 SOURCE=$5 ADOPTED=$6 CTL=$7
. /data/dev2/runs/dec/formal/m3-ops/m3-formal-lib.sh
S=/data/dev2/src/$SRC/src/training/decision2
D=/data/dev2/runs/dec/m3/formal-candidates/staging-${COMMIT:0:8}; mkdir -p $D
export HF_HUB_CACHE=/data/dev2/hf-cache
/data/dev2/tools/hf-cli/bin/hf download llm-semantic-router/dev2-dec-staging --revision $COMMIT --include "m3/$NAME/*" --local-dir $D \
  > $D/$NAME.download.log 2>&1 || { flog "$NAME download FAILED"; exit 1; }
(cd $D && sha256sum -c --quiet /data/dev2/runs/dec/m3/formal-candidates/$NAME.nodeB.sha256 > $D/$NAME.verify.log 2>&1) \
  && flog "$NAME download verified ($(wc -l < /data/dev2/runs/dec/m3/formal-candidates/$NAME.nodeB.sha256) files, staging $COMMIT)" \
  || { flog "$NAME hash verification FAILED"; exit 1; }
P=$D/m3/$NAME
case $TIER in 4B) PEER=/data/dev2/runs/eval/m1-adopt/decider4b; PN=decider4b ;; 2B) PEER=/data/dev2/runs/eval/m1-adopt/decider2b; PN=decider2b ;; *) PEER=; PN= ;; esac
spec=$S/v2/dec/adapter-spec-infer-dec.json
common=(--extra source=$SOURCE --extra model_id=decision2-dec-m3-$NAME --extra max_length=16384 --extra calibration=$P/cal698-16k/calibration.json --mount $D)
collect $SRC m3-$NAME-nodeA $P/checkpoint m3-$NAME-$COMMIT $spec "M3 formal post-key collection $NAME (prereg 5c8bbc569)" "${common[@]}" || exit 1
cmp=($ADOPTED adopted-1.0 $CTL same-limit-16k)
[ -n "$PEER" ] && cmp+=($PEER $PN)
score $SRC m3-$NAME-nodeA "decision2-dec-m3-$NAME" $TIER decision2 $P/checkpoint "${cmp[@]}"
mlx $SRC m3-$NAME $P/checkpoint m3-$NAME-$COMMIT $spec "${common[@]}"
flog "$NAME formal chain done"
