for g in 0 1 2 3 4; do mkdir -p /data/dev2/leases/gpu$g.lock; printf 'track=dec\nstatus=busy\npurpose=M2 preregistered arms (prereg 143cfc214)\nstart_utc=%s\nexpected_end_utc=2026-09-28T14:30:00Z\n' $now > /data/dev2/leases/gpu$g.lock/owner; done

NOX=/hf/models--llm-semantic-router--Decision-1.0-Nox-4B/snapshots/cde2a68dbaa557ea65dc458104d410a0802ee259
