#!/usr/bin/env bash
# DEV2.0-2B private release run: build, examples, repeatability, card, full-panel parity, upload, real download,
# re-hash, post-download checks and Hub readback. Stops before the collection add (no --collect).
set -euo pipefail
TS=$(date -u +%Y%m%dT%H%M%SZ)
SRC=336ac2a5a2adf661aeb40d946004159ea8d8c9a9-src_training_decision2; S=/data/dev2/src/$SRC/src/training/decision2
G=/data/dev2/private/panels/goldfree; P=/data/dev2/runs/release/inputs/dev2-2b-t1/derived
FROZEN=/data/dev2/runs/dec/formal/m3/m3-S2T-soup-nodeA-triton
TC=/data/dev2/runs/release/triton/dev2-2b-S2T-soup-nodeA-copy-$TS
W=/data/dev2/runs/release/dev2-2b-release-$TS
cp -a $FROZEN $TC
digest() { (cd "$1" && find . -type f -print0 | sort -z | xargs -0 sha256sum | sha256sum | cut -c1-64); }
echo "cache copy $TC files=$(find $TC -type f | wc -l) digest=$(digest $TC) frozen_digest=$(digest $FROZEN)"
set -x
$S/v2/release/release.sh --spec $S/v2/release/specs/dev2-2b-release.json --src $SRC --work $W \
  --gpu 5 --track release --shared-lease release --threads 4 \
  --site /opt/decision-fla --require-kernels \
  --env HIP_FORCE_DEV_KERNARG=1 --env TRITON_CACHE_AUTOTUNING=1 --env TRITON_CACHE_DIR=$TC --mount-rw $TC \
  --mount $G --mount $P \
  --parity typed-final:$G/typed-final.prompts.jsonl:$P/typed-final.predictions.jsonl:1600 \
  --parity css15:$G/css15.prompts.jsonl:$P/css15.predictions.jsonl:6547 \
  --parity public231:$G/public231.prompts.jsonl:$P/public231.predictions.jsonl:231 \
  --parity mlx-diag:$G/mlx-diag.prompts.jsonl:$P/mlx-diag.predictions.jsonl:2275 \
  --upload
set +x
echo "cache after run files=$(find $TC -type f | wc -l) digest=$(digest $TC)"
