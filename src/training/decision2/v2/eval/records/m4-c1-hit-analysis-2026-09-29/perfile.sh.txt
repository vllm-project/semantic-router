#!/usr/bin/env bash
# usage: perfile.sh NODE PROTECTED_DIR SCANDIR   (17 training-type rows vs every training-type file, one label per file)
set -euo pipefail
umask 077
NODE=$1 PD=$2 D=$3
SRC=ecf312c2bcc9fb22b6dde383fc65c46ae77ac31c-src_training_decision2
S=/data/dev2/src/$SRC/src/training/decision2
IMG=sha256:f83b1d10f14dbe46ea14ee56fd3e5d01849673f3739fed5311c99ba54cbc2d54
O=/data/dev2/runs/eval/m4/c1-hitgroups/perfile-$NODE
mkdir -p "$O"
cd "$S"
python3 -m v2.eval.sealed.hitgroups subset --protected "$PD/flagged-protected.jsonl" --ids "$PD/train17-ids.txt" --output "$PD/train17-protected.jsonl"
python3 - "/data/dev2/runs/eval/m4/c1-hitgroups/$NODE/grouped-manifest.json" "$O/perfile-manifest.json" <<'EOF'
import json, re, sys
TRAIN = re.compile(r"^(training-data:|/data/dev2/runs/(?!eval)|/data/dev2/private/(data|27b|dec-arms|a7/runs|a7/base)|/data/dev2/tmp|hf-blobs)")
m = json.load(open(sys.argv[1]))
labels = {}
for group, entry in m["labels"].items():
    if TRAIN.match(group):
        for item in entry["files"]:
            if not __import__("os").path.exists(item["path"]):
                continue
            labels[f"{group}::{item['path']}"] = {"kind": "training", "files": [item]}
json.dump({"schema": "c1-corpora/1", "labels": labels}, open(sys.argv[2], "w"))
print(len(labels), "files", sum(e["files"][0]["bytes"] for e in labels.values()) / 1e9, "GB")
EOF
MOUNTS=()
while IFS= read -r dir; do MOUNTS+=(-v "$dir:$dir:ro"); done < <(python3 -c 'import json,sys; print("\n".join(json.load(open(sys.argv[1]))["roots"].values()))' "$D/coverage-receipt.json")
echo "$(date -u +%Y-%m-%dT%H:%M:%SZ) HITGROUPS 17 training-type flagged protected rows read (read-only) on $NODE for the analysis-only per-file scan" >>"$PD/ACCESS.log"
docker run --rm --network none -v "$S:$S:ro" "${MOUNTS[@]}" --mount "type=tmpfs,destination=/data/dev2/private/sealed" \
  -v "$PD:$PD:ro" -v "$D:$D:ro" -v "$O:$O" -e PYTHONPATH="$S" -e PYTHONDONTWRITEBYTECODE=1 -w "$S" \
  --entrypoint python3 "$IMG" -m v2.eval.sealed.overlap scan --protected "$PD/train17-protected.jsonl" --manifest "$O/perfile-manifest.json" \
  --no-verify --workers 48 --exact-min-tokens 8 --output "$O/overlap-receipt.json" --hits "$O/overlap-hits.jsonl" >"$O/scan.log" 2>&1
echo DONE >>"$O/scan.log"
